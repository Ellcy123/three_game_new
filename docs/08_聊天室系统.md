# 《三兄弟的冒险2》游戏策划文档（开发版 v2.0）

## 八、聊天室系统

### 8.1 功能概述

**功能名称：** 实时聊天室

**功能目标：** 让三位玩家在游戏过程中可以实时文字交流，增强社交体验和团队协作

**核心特性：**
- ✅ 实时同步：消息即时推送给所有房间成员
- ✅ 身份识别：显示玩家自定义名字和角色头像
- ✅ 全局可用：几乎所有游戏环节均可聊天
- ✅ 条件禁言：特定关卡自动禁止聊天（百变小鹦BOSS战）

### 8.2 聊天室数据结构

#### 8.2.1 消息数据结构

```typescript
interface ChatMessage {
    id: string;              // 消息唯一ID
    roomId: string;          // 房间ID

    senderName: string;      // 发送者名字（玩家自定义）
    senderAvatar: string;    // 发送者头像（角色图标）
    content: string;         // 消息内容
    timestamp: number;       // 发送时间戳
    type: 'chat' | 'system'; // 消息类型：聊天/系统通知
}
```

```typescript
interface PlayerInfo {
    id: number;              // 玩家ID (1/2/3)

    avatar: string;          // 头像（🐱/🐕/🐢）

```

#### 8.2.3 聊天状态

```typescript
interface ChatState {
    enabled: boolean;        // 聊天是否启用
```

```javascript
// 游戏阶段变化时检查聊天状态
function onGamePhaseChange(newPhase) {
    if (newPhase === 'BOSS_PARROT') {
        // 进入百变小鹦BOSS战
        disableChat({

        enableChat();

        // 如果是刚从百变小鹦出来
        if (previousPhase === 'BOSS_PARROT') {
            sendSystemMessage('💬 聊天已恢复！');
        }
    }
}
```

### 8.5 核心功能实现

#### 8.5.1 发送消息

```javascript
// 客户端发送消息
function sendChatMessage(content) {
    // 检查聊天是否启用
    if (!chatState.enabled) {
        showToast('当前阶段禁止聊天');
        return;
    }

    // 内容校验
    const trimmedContent = content.trim();
    if (trimmedContent.length === 0) return;
    if (trimmedContent.length > 200) {
        showToast('消息过长，请控制在200字以内');
        return;
    }

    // 发送到服务器
    socket.emit('chat:send', {
        content: trimmedContent
    });

    // 清空输入框
    clearInput();
}
```

#### 8.5.2 接收消息

```javascript
// 客户端监听消息
socket.on('chat:message', (message: ChatMessage) => {
    // 添加到消息列表

    if (isChatCollapsed) {

    if (message.senderId !== currentPlayer.id) {
```

#### 8.5.3 服务端广播

```javascript
// 服务端处理聊天消息
socket.on('chat:send', ({ content }) => {
    const room = getRoom(socket.roomId);
    const player = getPlayer(socket.playerId);

    // 检查聊天是否启用
    if (!room.chatEnabled) {
        socket.emit('chat:error', { message: '当前阶段禁止聊天' });
        return;
    }

    // 构造消息对象
    const message: ChatMessage = {
        id: generateId(),
        roomId: room.id,
        senderId: player.id,
        senderName: player.name,
        senderAvatar: player.avatar,
        content: content,
        timestamp: Date.now(),
        type: 'chat'
    };

    // 广播给房间内所有玩家
    io.to(room.id).emit('chat:message', message);

    // 保存到消息历史（可选）
    room.chatHistory.push(message);

    // 限制历史消息数量
    if (room.chatHistory.length > 100) {
        room.chatHistory.shift();
    }
});
```

#### 8.5.4 系统消息

```javascript
// 发送系统消息
function sendSystemMessage(roomId, content) {
    const message: ChatMessage = {

        senderAvatar: '📢',
        content: content,
        timestamp: Date.now(),
        type: 'system'

}
```

### 8.8 特殊情况处理

#### 8.8.1 玩家掉线

```javascript
// 玩家掉线时
function onPlayerDisconnect(player, room) {
    sendSystemMessage(room.id, `${player.avatar} ${player.name} 已掉线`);
}

// 玩家重连时
function onPlayerReconnect(player, room) {
    sendSystemMessage(room.id, `${player.avatar} ${player.name} 已重新连接`);

    // 发送历史消息给重连玩家
    socket.emit('chat:history', room.chatHistory.slice(-50));
}
```

#### 8.8.2 消息频率限制

```javascript
// 防刷屏机制
const rateLimiter = {
    lastMessageTime: 0,
    minInterval: 500,  // 最小发送间隔500ms

    canSend() {
        const now = Date.now();
        if (now - this.lastMessageTime < this.minInterval) {
            return false;
        }
        this.lastMessageTime = now;
        return true;
    }
};

function sendChatMessage(content) {
    if (!rateLimiter.canSend()) {
        showToast('发送太频繁，请稍后再试');
        return;
    }
    // ... 发送逻辑
}
```

#### 8.8.3 消息内容过滤

```javascript
// 基础内容过滤
function filterContent(content) {
    // 去除首尾空格
    let filtered = content.trim();

    // 限制长度
    if (filtered.length > 200) {
        filtered = filtered.substring(0, 200);
    }

    // 过滤HTML标签（防XSS）
    filtered = filtered.replace(/<[^>]*>/g, '');

    // 敏感词过滤（可选）
    // filtered = sensitiveWordFilter(filtered);

    return filtered;
}
```
